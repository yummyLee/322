## 1. 核心思路

**储物箱 = 一个挂在世界物体上的独立 Inventory + 持久化标识。**

- 复用现有 `Inventory`（RefCounted）做格子管理，不改任何已有代码
- 储物箱只是一个 `Inventory` 的宿主 + 交互入口 + 存档键
- 箱子类型 = `StorageProfile` Resource，决定容量/分页/过滤/外观

```
StorageChest (世界节点, Area2D)
  ├── StorageProfile (Resource)    → 定义箱子属性
  ├── Inventory (RefCounted)       → 实际物品存储（复用）
  └── 交互 → 打开 StoragePanel     → 玩家箱子 + 背包双栏 UI
```

## 2. StorageProfile Resource

### 2.1 字段设计

| 字段 | 类型 | 说明 |
|------|------|------|
| `chest_id` | String | 唯一标识，用于存档键（如 "town_stash_01"） |
| `display_name` | String | UI 标题（如 "城镇储物箱"） |
| `slot_count` | int | 总格子数，默认 20 |
| `tabs` | Array[StorageTabDef] | 分页定义（空 = 不分页，单页） |
| `allowed_types` | Array[int] | 允许放入的 ItemType（空 = 全部允许） |
| `allowed_rarities` | Array[int] | 允许放入的稀有度（空 = 全部允许） |
| `allow_deposit` | bool | 是否允许存入（默认 true） |
| `allow_withdraw` | bool | 是否允许取出（默认 true） |
| `persist_across_sessions` | bool | 是否存档持久化（默认 true） |
| `max_unique_items` | int | 最多容纳不同物品种类数（0 = 不限） |

### 2.2 StorageTabDef

| 字段 | 类型 | 说明 |
|------|------|------|
| `tab_name` | String | 分页名称（如 "武器", "材料"） |
| `icon_color` | Color | 分页标签颜色 |
| `slot_count` | int | 该分页的格子数 |
| `allowed_types` | Array[int] | 该分页过滤（覆盖箱子全局过滤） |

### 2.3 代码骨架

```gdscript
# scripts/world/storage_profile.gd
class_name StorageProfile
extends Resource

@export var chest_id: String = ""
@export var display_name: String = "储物箱"
@export var slot_count: int = 20
@export var tabs: Array[StorageTabDef] = []
@export var allowed_types: Array[int] = []       # ItemData.ItemType
@export var allowed_rarities: Array[int] = []     # ItemData.ItemRarity
@export var allow_deposit: bool = true
@export var allow_withdraw: bool = true
@export var persist_across_sessions: bool = true
@export var max_unique_items: int = 0


## 实际使用的总格子数（有分页取分页之和，无分页取 slot_count）
func total_slots() -> int:
	if tabs.is_empty():
		return slot_count
	var total := 0
	for tab in tabs:
		total += tab.slot_count
	return total


## 检查物品是否允许放入
func can_accept(item: ItemData) -> bool:
	if not allowed_types.is_empty():
		if not allowed_types.has(item.item_type):
			return false
	if not allowed_rarities.is_empty():
		if not allowed_rarities.has(item.rarity):
			return false
	return true


## 有分页时，返回物品应放入的分页索引；无分页返回 0
func tab_index_for(item: ItemData) -> int:
	for i in tabs.size():
		var tab := tabs[i]
		if tab.allowed_types.is_empty() or tab.allowed_types.has(item.item_type):
			return i
	return 0
```

## 3. StorageInventory 运行时

### 3.1 职责

包装 `Inventory`，增加箱子特有的逻辑：分页映射、过滤校验、存档序列化。

```gdscript
# scripts/world/storage_inventory.gd
class_name StorageInventory
extends RefCounted

signal changed(tab_index: int, slot_index: int)

var profile: StorageProfile
var chest_id: String = ""

## 无分页时 _inventories[0]；有分页时每页一个 Inventory
var _inventories: Array[Inventory] = []
var _tab_offset: Array[int] = []  # 每个分页在全局格子中的起始偏移


func setup(p: StorageProfile) -> void:
	profile = p
	chest_id = p.chest_id
	_inventories.clear()
	_tab_offset.clear()

	if p.tabs.is_empty():
		_inventories.append(_create_inventory(p.slot_count))
		_tab_offset.append(0)
	else:
		var offset := 0
		for tab in p.tabs:
			_inventories.append(_create_inventory(tab.slot_count))
			_tab_offset.append(offset)
			offset += tab.slot_count

	# 连接每个子 Inventory 的信号
	for i in _inventories.size():
		var inv := _inventories[i]
		inv.inventory_changed.connect(_on_sub_inventory_changed.bind(i))


func _create_inventory(slots: int) -> Inventory:
	var inv := Inventory.new()
	# Inventory._init() 已默认分配 SLOT_COUNT 个槽，使用 resize_slots 统一调整
	inv.resize_slots(slots)
	return inv


func get_inventory(tab_index: int = 0) -> Inventory:
	if tab_index < 0 or tab_index >= _inventories.size():
		return null
	return _inventories[tab_index]


func can_accept(item: ItemData) -> bool:
	return profile.can_accept(item)


func tab_index_for(item: ItemData) -> int:
	return profile.tab_index_for(item)


func get_slot(tab_index: int, slot_index: int) -> Variant:
	var inv := get_inventory(tab_index)
	if inv == null:
		return null
	return inv.get_slot(slot_index)


func add_item(item: ItemData, quantity: int = 1) -> bool:
	var tab_idx := tab_index_for(item)
	var inv := get_inventory(tab_idx)
	if inv == null:
		return false
	if not can_accept(item):
		return false
	return inv.add_item(item, quantity)


func remove_item(tab_index: int, slot_index: int, quantity: int = 1) -> void:
	var inv := get_inventory(tab_index)
	if inv == null:
		return
	inv.remove_item(slot_index, quantity)


## 存档序列化
func serialize() -> Dictionary:
	var data := {"chest_id": chest_id, "tabs": []}
	for inv in _inventories:
		var tab_data := []
		for slot in inv._slots:
			if slot == null:
				tab_data.append(null)
			else:
				tab_data.append({
					"item_id": slot["item"].id,
					"quantity": slot["quantity"],
					"durability": Inventory.slot_get_durability(slot),
				})
		data["tabs"].append(tab_data)
	return data


func deserialize(data: Dictionary) -> void:
	for ti in min(_inventories.size(), data.get("tabs", []).size()):
		var tab_data: Array = data["tabs"][ti]
		var inv := _inventories[ti]
		for si in min(inv._slots.size(), tab_data.size()):
			var entry = tab_data[si]
			if entry == null:
				inv._slots[si] = null
			else:
				var item := ItemDatabase.get_item(entry["item_id"])
				if item != null:
					var slot := {"item": item, "quantity": entry["quantity"]}
					if item.max_durability > 0:
						slot["durability"] = entry.get("durability", item.max_durability)
					inv._slots[si] = slot
				else:
					inv._slots[si] = null


## bind(i) 将 tab_index 追加为第二参数；signal 定义为 changed(tab_index, slot_index)
func _on_sub_inventory_changed(slot_index: int, tab_index: int) -> void:
	changed.emit(tab_index, slot_index)  # 顺序与 signal 定义匹配
```

## 4. StorageChest 世界节点

### 4.1 设计

```gdscript
# scripts/world/storage_chest.gd
class_name StorageChest
extends Area2D

signal chest_opened
signal chest_closed

@export var profile: StorageProfile
@export var persist_id: String = ""      # 存档键，默认用 profile.chest_id
@export var interaction_range: float = 48.0

var storage: StorageInventory = null
var _is_open: bool = false
var _player_nearby: Player = null


func _ready() -> void:
	if profile == null:
		push_error("StorageChest %s: 缺少 StorageProfile" % name)
		return

	if persist_id.is_empty():
		persist_id = profile.chest_id

	storage = StorageInventory.new()
	storage.setup(profile)

	# 从存档恢复
	if profile.persist_across_sessions:
		_load_from_save()

	body_entered.connect(_on_body_entered)
	body_exited.connect(_on_body_exited)


func _on_body_entered(body: Node) -> void:
	if body is Player:
		_player_nearby = body


func _on_body_exited(body: Node) -> void:
	if body is Player and _player_nearby == body:
		_player_nearby = null
		if _is_open:
			close()


func try_open(player: Player) -> bool:
	if _is_open:
		return false
	if player == null:
		return false
	var dist := global_position.distance_to(player.global_position)
	if dist > interaction_range:
		return false
	_is_open = true
	chest_opened.emit()
	UILayerManager.open_storage(self, player)
	return true


func close() -> void:
	if not _is_open:
		return
	_is_open = false
	if profile.persist_across_sessions:
		_save_to_save()
	chest_closed.emit()
	UILayerManager.close_storage()


func _save_to_save() -> void:
	SaveManager.set_storage_data(persist_id, storage.serialize())


func _load_from_save() -> void:
	var data := SaveManager.get_storage_data(persist_id)
	if not data.is_empty():
		storage.deserialize(data)
```

## 5. 交互流程

```
玩家靠近 StorageChest
  ├── body_entered → _player_nearby = player
  ├── 玩家按 E / 交互键 → Player 调 chest.try_open()
  │     ├── 距离校验
  │     ├── UILayerManager.open_storage(chest, player)
  │     │     ├── 创建 StoragePanel 场景
```

## 6. SaveManager 集成

在 `SaveManager` 的 `_world_state` 中新增存储分区：

```gdscript
# SaveManager 新增方法（注意：实际已实现，此处供参考）

## 所有储物箱数据 { "chest_id": { ... } }
## 注意：Godot 4 Dictionary 无 get_or_add()，需先判断键是否存在
func set_storage_data(chest_id: String, data: Dictionary) -> void:
	if not _world_state.has("storage_chests") or typeof(_world_state["storage_chests"]) != TYPE_DICTIONARY:
		_world_state["storage_chests"] = {}
	(_world_state["storage_chests"] as Dictionary)[chest_id] = data.duplicate(true)

func get_storage_data(chest_id: String) -> Dictionary:
	var chests: Variant = _world_state.get("storage_chests", {})
	if typeof(chests) != TYPE_DICTIONARY:
		return {}
	var raw: Variant = (chests as Dictionary).get(chest_id, {})
	return (raw as Dictionary).duplicate(true) if typeof(raw) == TYPE_DICTIONARY else {}

func clear_storage_data(chest_id: String) -> void:
	if not _world_state.has("storage_chests"):
		return
	var chests: Dictionary = _world_state["storage_chests"] as Dictionary
	chests.erase(chest_id)
```

### 7.4 拆分操作（分堆）

拆分是储物箱最常见的痛点，设计两种路径：

#### 路径 A：Shift + 左键（弹窗输入）
```
玩家 Shift+左键 点击"木材 ×64"
  └─ 弹出数量输入弹窗
        ├── 输入框（默认值 = 总数量的一半，向上取整）
        ├── [-] [+] 按钮（每次 ±1）
        ├── [½] 按钮（快速填入一半）
        ├── [Max] 按钮（填入全部）
        └── [确认] → 悬浮指定数量在光标 / [取消]
```

#### 路径 B：右键连续点放（精细控制）
```
右键拾起半堆（32个）→ 光标持有 32
  └─ 在目标格上每次右键放下 1 个
       └─ 重复直到放完或按左键放下剩余全部
```

#### 数量输入框设计要点
- 支持键盘直接输入数字
- `Enter` 确认，`Esc` 取消
- 数值范围限定 `[1, slot.quantity]`，自动 clamp
- 非法字符输入时不回显（只接受数字）

### 7.5 排序功能

#### 7.5.1 排序入口

| 入口 | 位置 | 说明 |
|------|------|------|
| 排序按钮（带下拉菜单） | 面板顶栏 | 点击展开排序方式选择 |
| `Ctrl + S` 快捷键 | 任意时刻 | 使用上次选择的排序方式 |
| 右键菜单 → 排序 | 右键空白格 | 上下文菜单触发 |

#### 7.5.2 排序维度

| 排序方式 | 优先级逻辑 | 典型游戏 |
|---------|-----------|---------|
| **按类型** | 武器 > 防具 > 饰品 > 消耗品 > 材料 > 其他 | Diablo 4 |
| **按稀有度** | 传说 > 史诗 > 稀有 > 普通（降序） | PoE / WoW |
| **按名称** | 字母/拼音升序 | Minecraft |
| **按数量** | 数量多 → 少（堆叠物品先） | Terraria |
| **按重量** | 重 → 轻 或 轻 → 重 | — |
| **按最近获得** | 最新拾取的排前面 | V Rising |
| **按价值** | 金币价值 高 → 低 | — |
| **整理压缩** | 不改顺序，只合并同类堆叠、消除碎片空格 | Stardew Valley |

> **复合排序**：先按类型分组，组内再按稀有度 → 名称。这是大多数玩家最期望的默认行为。

#### 7.5.3 StorageInventory 排序接口

```gdscript
## 排序模式枚举（ui 层传给 StorageInventory）
enum SortMode {
	BY_TYPE,          # 按物品类型分组
	BY_RARITY,        # 按稀有度降序
	BY_NAME,          # 按名称字母升序
	BY_QUANTITY,      # 按数量降序
	BY_VALUE,         # 按金币价值降序
	COMPACT,          # 整理压缩（合并堆叠 + 清空碎片）
}

## 对指定分页（或全部，tab_index=-1）执行排序
func sort_tab(tab_index: int, mode: SortMode) -> void:
	var targets: Array[int] = []
	if tab_index < 0:
		for i in _inventories.size():
			targets.append(i)
	else:
		targets.append(tab_index)
	for ti in targets:
		var inv := _inventories[ti]
		_sort_inventory(inv, mode)
	# 排序完成后批量通知 UI 刷新
	for ti in targets:
		var inv := _inventories[ti]
		for si in inv.get_slot_count():
			changed.emit(ti, si)


func _sort_inventory(inv: Inventory, mode: SortMode) -> void:
	# 1. 收集非空格子
	var items: Array = []
	for slot in inv._slots:
		if slot != null:
			items.append(slot.duplicate())
	# 2. COMPACT：先合并同种堆叠
	if mode == SortMode.COMPACT:
		items = _merge_stacks(items)
	# 3. 排序
	match mode:
		SortMode.BY_TYPE:
			items.sort_custom(func(a, b): return _cmp_type_rarity_name(a, b))
		SortMode.BY_RARITY:
			items.sort_custom(func(a, b): return (a["item"] as ItemData).rarity > (b["item"] as ItemData).rarity)
		SortMode.BY_NAME:
			items.sort_custom(func(a, b): return (a["item"] as ItemData).item_name < (b["item"] as ItemData).item_name)
		SortMode.BY_QUANTITY:
			items.sort_custom(func(a, b): return int(a["quantity"]) > int(b["quantity"]))
		SortMode.BY_VALUE:
			items.sort_custom(func(a, b): return (a["item"] as ItemData).value_gold > (b["item"] as ItemData).value_gold)
		SortMode.COMPACT:
			items.sort_custom(func(a, b): return _cmp_type_rarity_name(a, b))
	# 4. 写回（先清空，再顺序填入，末尾补 null）
	for i in inv.get_slot_count():
		if i < items.size():
			inv._slots[i] = items[i]
		else:
			inv._slots[i] = null


func _cmp_type_rarity_name(a: Dictionary, b: Dictionary) -> bool:
	var ia := a["item"] as ItemData
	var ib := b["item"] as ItemData
	if ia.item_type != ib.item_type:
		return int(ia.item_type) < int(ib.item_type)
	if ia.rarity != ib.rarity:
		return int(ia.rarity) > int(ib.rarity)  # 稀有度高的在前
	return ia.item_name < ib.item_name


## 合并同种物品的堆叠（COMPACT 专用）
func _merge_stacks(items: Array) -> Array:
	var merged: Dictionary = {}  # item_id → {item, quantity, durability}
	for slot in items:
		var item := slot["item"] as ItemData
		if item.max_durability > 0 or item.max_stack <= 1:
			# 耐久品/不可堆叠 → 不合并
			if not merged.has(item.id + "_nodur"):
				merged[item.id + "_nodur"] = []
			(merged[item.id + "_nodur"] as Array).append(slot)
		else:
			if not merged.has(item.id):
				merged[item.id] = {"item": item, "quantity": 0}
			(merged[item.id] as Dictionary)["quantity"] += int(slot["quantity"])
	# 展开合并结果（超出 max_stack 的拆成多格）
	var result: Array = []
	for key in merged:
		var val = merged[key]
		if val is Array:
			result.append_array(val as Array)
		else:
			var d := val as Dictionary
			var item := d["item"] as ItemData
			var rem := int(d["quantity"])
			while rem > 0:
				var take := mini(rem, item.max_stack)
				result.append({"item": item, "quantity": take})
				rem -= take
	return result
```

### 7.6 批量操作

#### 7.6.1 全部存入 / 全部取出

```
[全部存入] 按钮（或快捷键 Shift+Ctrl+右键）
  └─ 遍历背包，将所有通过 profile.can_accept() 的物品存入箱子
       ├── 满格时停止，UI 上弹提示"箱子已满"
       └── 跳过锁定格（Alt+左键 标记的格子）

[全部取出] 按钮
  └─ 遍历箱子，将所有物品转移到背包
       └─ 背包满时停止，已转移的不回滚
```

#### 7.6.2 多选批量操作

```
Ctrl + 左键（不拖拽）→ 多选模式
  ├── 点击格子：加入/移出选中集合（高亮显示）
  ├── Shift + 左键：区域框选（起点到终点之间所有格子）
  └── 选中后可执行：
        ├── Enter / 右键菜单「转移选中」→ 批量转移到对面栏
        ├── Del / 右键菜单「丢弃」→ 弹确认弹窗，批量丢弃
        └── Esc → 取消选中，回到正常模式
```

#### 7.6.3 同类全选

```
双击一个格子（不触发转移的场合，如按住 Ctrl 时）
  └─ 同栏内所有相同 item_id 的格子进入选中状态
       └─ 方便一键转移所有同类物品
```
